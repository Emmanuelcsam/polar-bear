import os
from pathlib import Path
import torch
from torch.utils.data import Dataset

class FiberOpticsDatasetCustom(Dataset):
    """Modified dataset that handles non-standard directory structure."""
    
    def __init__(self, pt_dir, image_dir, mode='train'):
        self.pt_dir = Path(pt_dir)
        self.image_dir = Path(image_dir)
        self.mode = mode
        
        # Collect all .pt files
        self.pt_files = list(self.pt_dir.rglob("*.pt"))
        print(f"Found {len(self.pt_files)} .pt files")
        
        # Create label mapping based on filename patterns
        self.labels = self._infer_labels()
        
    def _infer_labels(self):
        """Infer labels from filenames or use default labeling."""
        labels = []
        for pt_file in self.pt_files:
            # Try to infer from filename
            filename = pt_file.name.lower()
            if 'core' in filename:
                labels.append(0)
            elif 'cladding' in filename:
                labels.append(1)
            elif 'ferrule' in filename:
                labels.append(2)
            elif 'defect' in filename:
                labels.append(3)
            else:
                # Default label - you may want to modify this
                labels.append(0)
        return labels
    
    def __len__(self):
        return len(self.pt_files)
    
    def __getitem__(self, idx):
        # Load tensor
        tensor = torch.load(self.pt_files[idx])
        label = self.labels[idx]
        return tensor, label
EOF_DATASET
