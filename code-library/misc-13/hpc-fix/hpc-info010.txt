onintr save_checkpoint_and_exit

==============================================================================
Launch Training
==============================================================================
echo " "
echo "========================================================"
echo "Starting fiber optics neural network training..."
echo "========================================================"

Check if resuming from checkpoint
set latest_checkpoint = ls -t $checkpoint_dir/checkpoint_epoch_*.pth |& head -1
if ("$latest_checkpoint" != "") then
echo "Found checkpoint: $latest_checkpoint"
set resume_flag = "--checkpoint $latest_checkpoint"
else
echo "Starting fresh training (no checkpoint found)"
set resume_flag = ""
endif

Launch training with srun for proper task distribution
if ($SLURM_NNODES > 1) then

Distributed training across multiple nodes
srun --ntasks=$SLURM_NTASKS --ntasks-per-node=20 

python $python_script 

--config config/config.yaml 

--config-override config/hpc_overrides.yaml 

$resume_flag
else

Single node training
python $python_script 

--config config/config.yaml 

--config-override config/hpc_overrides.yaml 

$resume_flag
endif

Save exit status
set exit_status = $status

==============================================================================
Post-Processing
==============================================================================
echo " "
echo "Training completed with exit status: $exit_status"

Copy results back to job directory
echo "Copying results back to job directory..."
rsync -av $checkpoint_dir/ $job_dir/checkpoints/
rsync -av $results_dir/ $job_dir/results/
rsync -av $logs_dir/ $job_dir/logs/

Copy the best model specifically
if (-f $checkpoint_dir/best_model.pth) then
cp $checkpoint_dir/best_model.pth $job_dir/checkpoints/
echo "Best model saved to: $job_dir/checkpoints/best_model.pth"
endif

echo " "
echo "========================================================"
echo "Job finished on: date"
echo "========================================================"

exit $exit_status

==============================================================================
Signal Handler Implementation
==============================================================================
save_checkpoint_and_exit:
echo " "
echo "Received termination signal, initiating checkpoint save..."

Send checkpoint signal to Python process
killall -USR1 python

Wait for checkpoint to complete
sleep 60

Copy current checkpoints back
rsync -av $checkpoint_dir/ $job_dir/checkpoints/

Check if we should resubmit
if ($exit_status != 0) then
echo "Resubmitting job to continue training..."

Create resubmission script
cat > resubmit_job.sh << 'RESUBMIT_EOF'
#!/bin/bash
cd $job_dir
sbatch run-job.slurm
RESUBMIT_EOF

Submit continuation job
set new_job = sbatch --parsable --dependency=afterany:$SLURM_JOB_ID resubmit_job.sh
echo "Resubmitted as job: $new_job"
endif

exit 0
