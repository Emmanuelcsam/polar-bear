#!/usr/bin/env python3
# =============================================================================
#  Fiber-Optic End-Face Defect Pipeline v2
#  Comprehensive D2L-style implementation that integrates:
#    • Extensive statistics (reports 001–014) for parameterization
#    • Dataset imbalance handling (dynamic class-weights + focal-loss)
#    • Advanced data augmentation, mix-up, cosine-annealing scheduler
#    • ResNeXt-34 backbone (grouped convs) as recommended by D2L §8.6
#    • Region-wise classification + blob / scratch defect metrics
#    • Full training loop & inference sweep through ./dataset/**
# -----------------------------------------------------------------------------
#  ALL helper functions, training utilities, and network blocks are adapted
#  verbatim or with minimal refactoring from code listings in https://d2l.ai/
# =============================================================================

import os, sys, glob, csv, math, json, random, time, itertools
from pathlib import Path
from collections import defaultdict, Counter

import cv2
import numpy as np
import torch
from torch import nn
from torch.utils.data import DataLoader, Dataset
from torchvision import transforms
from d2l import torch as d2l                         # D2L helpers

# -----------------------------------------------------------------------------#
# Global Hyper-parameters (tuned using attachment statistics)                   #
# -----------------------------------------------------------------------------#
class Args(d2l.HyperParameters):
    def __init__(self):
        # I/O
        self.root             = Path('.')
        self.data_dir         = self.root/'dataset'
        self.ref_dir          = self.root/'reference'
        self.ckpt_dir         = self.root/'checkpoints'
        self.stats_csv        = self.root/'defect_stats.csv'
        # Training
        self.batch_size       = 128
        self.num_epochs       = 50
        self.lr               = 1e-3
        self.min_lr           = 1e-5
        self.num_workers      = 4
        self.mixup_alpha      = 0.1
        self.label_smoothing  = 0.0
        self.device           = d2l.try_gpu()
        # Model
        self.groups           = 32        # ResNeXt cardinality
        self.bot_mul          = 1         # Bottleneck ratio (ResNeXt §8.6)
        self.dropout_rates    = (0.2,0.3,0.4)
        self.seed             = 42
args = Args()
torch.manual_seed(args.seed); random.seed(args.seed); np.random.seed(args.seed)
args.ckpt_dir.mkdir(exist_ok=True)

# -----------------------------------------------------------------------------#
# 1. Reference Tensor Dataset (supervised region classifier)                   #
# -----------------------------------------------------------------------------#
class RefTensorDataset(Dataset):
    def __init__(self, root:Path, aug=False):
        self.files, self.labels = [], []
        self.classes = sorted([p.name for p in root.iterdir() if p.is_dir()])
        self.cls2idx = {c:i for i,c in enumerate(self.classes)}
        for cls in self.classes:
            for f in (root/cls).glob('*.pt'):
                self.files.append(f)
                self.labels.append(self.cls2idx[cls])
        self.tfms = transforms.Compose([
            transforms.RandomHorizontalFlip(),
            transforms.RandomVerticalFlip(),
        ]) if aug else transforms.Compose([])
    def __len__(self):  return len(self.files)
    def __getitem__(self, idx):
        x = torch.load(self.files[idx]).float()          # (C,H,W) tensor
        if self.tfms: x = self.tfms(x)
        return x, torch.tensor(self.labels[idx])

ref_train = RefTensorDataset(args.ref_dir, aug=True)
ref_val   = RefTensorDataset(args.ref_dir, aug=False)

train_dl  = DataLoader(ref_train, batch_size=args.batch_size, shuffle=True,
                       num_workers=args.num_workers, pin_memory=True)
val_dl    = DataLoader(ref_val,   batch_size=args.batch_size, shuffle=False,
                       num_workers=args.num_workers, pin_memory=True)

# -----------------------------------------------------------------------------#
# 2. Compute dynamic class-weights (tackle heavy imbalance)                    #
# -----------------------------------------------------------------------------#
cls_counts = Counter(ref_train.labels)
weights = torch.tensor([1.0/cls_counts[c] for c in sorted(cls_counts)], 
                       dtype=torch.float, device=args.device)
weights /= weights.sum()                     # normalize
# -----------------------------------------------------------------------------#
# 3. Focal Loss implementation (D2L §4.6 + class-weights)                      #
# -----------------------------------------------------------------------------#
class FocalLoss(nn.Module):
    def __init__(self, alpha=None, gamma=2.0):
        super().__init__(); self.alpha = alpha; self.gamma = gamma
    def forward(self, logits, targets):
        ce = nn.functional.cross_entropy(logits, targets, reduction='none',
                                         weight=self.alpha)
        pt = torch.exp(-ce)
        return ((1-pt)**self.gamma * ce).mean()

# -----------------------------------------------------------------------------#
# 4. ResNeXt-34 Backbone (adapted from D2L §8.6 – ResNeXtBlock listing)        #
# -----------------------------------------------------------------------------#
class ResNeXtBlock(nn.Module): #@save from D2L
    def __init__(self, channels, groups, bot_mul, use_1x1conv=False, stride=1):
        super().__init__()
        bot_channels = int(round(channels*bot_mul))
        self.conv1 = nn.Conv2d(channels, bot_channels, 1)
        self.conv2 = nn.Conv2d(bot_channels, bot_channels, 3, stride,
                               padding=1, groups=bot_channels//groups)
        self.conv3 = nn.Conv2d(bot_channels, channels, 1)
        self.bn1 = nn.BatchNorm2d(bot_channels)
        self.bn2 = nn.BatchNorm2d(bot_channels)
        self.bn3 = nn.BatchNorm2d(channels)
        if use_1x1conv:
            self.conv4 = nn.Conv2d(channels, channels, 1, stride)
            self.bn4   = nn.BatchNorm2d(channels)
        else: self.conv4 = None
    def forward(self, X):
        Y = nn.functional.relu(self.bn1(self.conv1(X)))
        Y = nn.functional.relu(self.bn2(self.conv2(Y)))
        Y = self.bn3(self.conv3(Y))
        if self.conv4: X = self.bn4(self.conv4(X))
        return nn.functional.relu(Y+X)

def resnext34(num_classes,len_ref_classes):
    def block(num_blocks, channels, first):
        blk=[]
        for i in range(num_blocks):
            blk.append(ResNeXtBlock(channels,args.groups,args.bot_mul,
                                    use_1x1conv=not first and i==0,
                                    stride=2 if (not first and i==0) else 1))
        return nn.Sequential(*blk)
    net = nn.Sequential(
        nn.Conv2d(1,64,7,2,3), nn.BatchNorm2d(64), nn.ReLU(),
        nn.MaxPool2d(3,2,1),
        block(3,64,  True),
        block(4,128,False),
        block(6,256,False),
        block(3,512,False),
        nn.AdaptiveAvgPool2d((1,1)), nn.Flatten(),
        nn.Dropout(args.dropout_rates[-1]),
        nn.Linear(512,num_classes))
    return net

# -----------------------------------------------------------------------------#
# 5. D2L-style Classifier wrapper                                              #
# -----------------------------------------------------------------------------#
class RegionClassifier(d2l.Classifier):
    def __init__(self):
        super().__init__()
        self.save_hyperparameters()
        self.net = resnext34(len(ref_train.classes),len(ref_train.classes))
        self.net.apply(d2l.init_cnn)
        self.loss_fn = FocalLoss(alpha=weights, gamma=2.0)
    def forward(self, X): return self.net(X)

model = RegionClassifier().to(args.device)

# Optimizer + scheduler (cosine annealing)  ----------------------------------#
optimizer = torch.optim.AdamW(model.parameters(), lr=args.lr, weight_decay=1e-4)
scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(
    optimizer, T_max=args.num_epochs, eta_min=args.min_lr)
trainer = d2l.Trainer(max_epochs=args.num_epochs, num_gpus=1, gradient_clip_val=1.0)

# -----------------------------------------------------------------------------#
# 6. Mix-Up utility (D2L §14.1 image augmentation)                             #
# -----------------------------------------------------------------------------#
def mixup_data(x, y, alpha=1.0):
    if alpha>0.0:
        lam = np.random.beta(alpha, alpha)
    else: lam = 1.0
    batch_size = x.size()[0]
    index = torch.randperm(batch_size, device=x.device)
    mixed_x = lam * x + (1 - lam) * x[index, :]
    y_a, y_b = y, y[index]
    return mixed_x, y_a, y_b, lam

def mixup_criterion(pred, y_a, y_b, lam, loss_fn):
    return lam*loss_fn(pred, y_a) + (1 - lam)*loss_fn(pred, y_b)

# Override training step to include mix-up                                       #
@trainer.add_to_class
def training_step(self, batch):
    X, y = [a.to(args.device) for a in batch]
    if args.mixup_alpha>0:
        X, y_a, y_b, lam = mixup_data(X, y, args.mixup_alpha)
    self.optimizer.zero_grad()
    y_hat = self.model(X)
    if args.mixup_alpha>0:
        loss = mixup_criterion(y_hat, y_a, y_b, lam, self.model.loss_fn)
    else: loss = self.model.loss_fn(y_hat, y)
    loss.backward(); self.optimizer.step()
    return float(loss.detach())

trainer.optimizer = optimizer

# -----------------------------------------------------------------------------#
# 7. Train the regional classifier                                             #
# -----------------------------------------------------------------------------#
if not (args.ckpt_dir/'resnext34.pth').exists():
    trainer.fit(model, train_dl, val_dl, log_interval=20)
    torch.save(model.state_dict(), args.ckpt_dir/'resnext34.pth')
else:
    model.load_state_dict(torch.load(args.ckpt_dir/'resnext34.pth',
                                     map_location=args.device))
model.eval(); scheduler.step()

# -----------------------------------------------------------------------------#
# 8. Image Preprocessing / Region Isolation                                    #
# -----------------------------------------------------------------------------#
to_tensor = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize([0.5], [0.5])
])

def isolate_regions(gray):
    h,w = gray.shape
    blr  = cv2.medianBlur(gray,5)
    circles = cv2.HoughCircles(blr,cv2.HOUGH_GRADIENT,1.2,h//8,
                               param1=50,param2=30,minRadius=h//6,maxRadius=h//2)
    masks={}
    if circles is not None and len(circles[0])>=2:
        circles = sorted(np.uint16(circles[0]), key=lambda c:c[2])
        core_r, clad_r = circles[0][2], circles[1][2]
        cx,cy = circles[0][:2]
        Y,X = np.ogrid[:h,:w]
        dist = np.sqrt((X-cx)**2+(Y-cy)**2)
        masks['core']     = (dist<=core_r ).astype(np.uint8)*255
        masks['cladding'] = ((dist>core_r)&(dist<=clad_r)).astype(np.uint8)*255
        masks['ferrule']  = (dist>clad_r).astype(np.uint8)*255
    return masks

# Blob detector parameters (from OpenCV docs, tuned with report stats) --------#
blob_params = cv2.SimpleBlobDetector_Params()
blob_params.filterByArea=True; blob_params.minArea=20
blob_detector = cv2.SimpleBlobDetector_create(blob_params)

def detect_blobs(masked):
    return len(blob_detector.detect(masked))

def detect_scratches(masked):
    sob  = cv2.Sobel(masked,cv2.CV_64F,1,0,ksize=3)+cv2.Sobel(masked,cv2.CV_64F,0,1,ksize=3)
    edges = cv2.convertScaleAbs(sob)
    cnts,_ = cv2.findContours(edges,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
    return sum(cv2.arcLength(c,False)>50 for c in cnts)

# -----------------------------------------------------------------------------#
# 9. Inference sweep through ./dataset/**                                      #
# -----------------------------------------------------------------------------#
stats = defaultdict(Counter)
img_files = sorted(itertools.chain(
    glob.glob(str(args.data_dir/'**'/'*.png'), recursive=True),
    glob.glob(str(args.data_dir/'**'/'*.jpg'), recursive=True)))

t0=time.time()
for fp in img_files:
    img = cv2.imread(fp,cv2.IMREAD_COLOR)
    gray = cv2.cvtColor(img,cv2.COLOR_BGR2GRAY)
    masks = isolate_regions(gray)
    for region,mask in masks.items():
        roi = cv2.bitwise_and(gray,gray,mask=mask)
        tensor = to_tensor(roi).unsqueeze(0).to(args.device)
        with torch.no_grad():
            pred = torch.argmax(model(tensor),1).item()
        cls_name = ref_train.classes[pred]
        stats[region]['images']+=1
        stats[region]['blobs']+=detect_blobs(roi)
        stats[region]['scratches']+=detect_scratches(roi)
        stats[region][cls_name]+=1
t1=time.time()

# -----------------------------------------------------------------------------#
# 10. Save CSV summary                                                         #
# -----------------------------------------------------------------------------#
header = ['region','images','blobs','scratches']+ref_train.classes
with open(args.stats_csv,'w',newline='') as fp:
    writer = csv.writer(fp); writer.writerow(header)
    for region,ctr in stats.items():
        row=[region,ctr['images'],ctr['blobs'],ctr['scratches']]
        row+=[ctr[c] for c in ref_train.classes]; writer.writerow(row)

print('=== Inspection Summary ================================================')
for reg,ctr in stats.items():
    print(f"{reg.upper():10} | imgs {ctr['images']:5} | blobs {ctr['blobs']:4} | "
          f"scratches {ctr['scratches']:4}")
print(f'Processed {len(img_files)} images in {t1-t0:.1f}s')
print(f'Saved statistics → {args.stats_csv.resolve()}')
